```python
"""In-memory Firestore stub.

Simulates Firestore collections and documents with no GCP credentials required.
Interface matches the access patterns used by IncidentMemoryAgent and PostmortemAgent.
"""

from copy import deepcopy
from typing import Any, Optional


class FirestoreStub:
    """In-memory Firestore replacement for local development.
    
    Supports:
    - get_document(collection, doc_id) → dict | None
    - set_document(collection, doc_id, data) → None
    - delete_document(collection, doc_id) → None
    - list_collection(collection) → list[dict]
    - query_collection(collection, field, value) → list[dict]
    """

    def __init__(self) -> None:
        # Structure: {collection_name: {doc_id: data_dict}}
        self._data: dict[str, dict[str, dict]] = {}

    def get_document(self, collection: str, doc_id: str) -> Optional[dict]:
        """Return a document by collection and ID, or None if not found."""
        return deepcopy(self._data.get(collection, {}).get(doc_id))

    def set_document(self, collection: str, doc_id: str, data: dict[str, Any]) -> None:
        """Write or overwrite a document."""
        self._data.setdefault(collection, {})[doc_id] = deepcopy(data)

    def delete_document(self, collection: str, doc_id: str) -> None:
        """Delete a document. No-op if not found."""
        self._data.get(collection, {}).pop(doc_id, None)

    def list_collection(self, collection: str) -> list[dict]:
        """Return all documents in a collection."""
        return [deepcopy(doc) for doc in self._data.get(collection, {}).values()]

    def query_collection(
        self, collection: str, field: str, value: Any
    ) -> list[dict]:
        """Return documents where field equals value."""
        return [
            deepcopy(doc)
            for doc in self._data.get(collection, {}).values()
            if doc.get(field) == value
        ]

    def clear(self) -> None:
        """Clear all data. Useful for test isolation."""
        self._data.clear()
```

```python
"""Cloud Monitoring stub returning canned metric snapshots."""

from copy import deepcopy
from datetime import datetime, timezone

from deployguard.stubs.fixtures.monitoring import BASELINE_METRICS, MONITORING_FIXTURES


class MonitoringStub:
    """Stub for Google Cloud Monitoring.
    
    Returns deterministic fixture metrics keyed by scenario name.
    Use deployment_id as the scenario key in local dev.
    """

    def get_metrics(self, deployment_id: str, window_minutes: int = 5) -> dict:
        """Return metric snapshot for deployment_id.
        
        If deployment_id matches a fixture scenario key, return that scenario.
        Otherwise return the healthy baseline.
        """
        scenario = deployment_id if deployment_id in MONITORING_FIXTURES else "healthy"
        metrics = deepcopy(MONITORING_FIXTURES[scenario])
        metrics["deployment_id"] = deployment_id
        metrics["window_minutes"] = window_minutes
        metrics["collected_at"] = datetime.now(tz=timezone.utc).isoformat()
        return metrics

    def get_baseline(self, service_name: str) -> dict:
        """Return the pre-deployment baseline metrics for a service."""
        baseline = deepcopy(BASELINE_METRICS)
        baseline["service_name"] = service_name
        return baseline

    def list_available_scenarios(self) -> list[str]:
        """Return available scenario keys (useful for testing)."""
        return list(MONITORING_FIXTURES.keys())
```

```python
"""Cloud Logging stub returning canned log entries."""

from copy import deepcopy
from datetime import datetime, timezone

from deployguard.stubs.fixtures.logging import LOGGING_FIXTURES


class LoggingStub:
    """Stub for Google Cloud Logging.
    
    Returns deterministic log entries keyed by scenario name.
    Includes the prompt injection fixture for Phase 6 security demo.
    """

    def get_logs(
        self,
        deployment_id: str,
        limit: int = 100,
        severity_filter: str | None = None,
    ) -> list[dict]:
        """Return log entries for deployment_id.
        
        If deployment_id matches a fixture scenario key, return that scenario.
        Otherwise return healthy logs.
        """
        scenario = deployment_id if deployment_id in LOGGING_FIXTURES else "healthy"
        logs = deepcopy(LOGGING_FIXTURES[scenario])
        
        # Add metadata to each log entry
        now = datetime.now(tz=timezone.utc).isoformat()
        for entry in logs:
            entry.setdefault("timestamp", now)
            entry.setdefault("deployment_id", deployment_id)
        
        # Apply severity filter if provided
        if severity_filter:
            logs = [e for e in logs if e.get("severity") == severity_filter]
        
        return logs[:limit]

    def list_available_scenarios(self) -> list[str]:
        """Return available scenario keys."""
        return list(LOGGING_FIXTURES.keys())
```

```python
        stub = FirestoreStub()

        stub = FirestoreStub()
        data = {"incident_id": "INC-001", "severity": "HIGH"}
        stub.set_document("incidents", "INC-001", data)
        result = stub.get_document("incidents", "INC-001")

        """Mutations to returned dict should not affect stored data."""
        stub = FirestoreStub()

        result = stub.get_document("items", "A")
```
